**src/ui/MarketScreen.jsx**
```jsx
import React, { useState } from 'react';
import { View, TextInput, Text, TouchableOpacity, FlatList, StyleSheet } from 'react-native';
import './Theme.css';

const dummyStocks = [
  { symbol: 'AAPL', name: 'Apple Inc.', price: 172.34 },
  { symbol: 'TSLA', name: 'Tesla, Inc.', price: 215.12 },
  { symbol: 'BTCUSD', name: 'Bitcoin', price: 31745.6 },
  // Add more as needed
];

export default function MarketScreen({ navigation }) {
  const [search, setSearch] = useState('');
  const filtered = dummyStocks.filter(
    (s) => s.symbol.includes(search.toUpperCase()) || s.name.toLowerCase().includes(search.toLowerCase())
  );

  const renderItem = ({ item }) => (
    <TouchableOpacity
      style={styles.stockRow}
      onPress={() => navigation.navigate('StockDetail', { symbol: item.symbol })}
    >
      <Text style={styles.symbol}>{item.symbol}</Text>
      <Text style={styles.name}>{item.name}</Text>
      <Text style={styles.price}>${item.price.toFixed(2)}</Text>
    </TouchableOpacity>
  );

  return (
    <View style={styles.container}>
      <Text style={styles.header}>Market</Text>
      <TextInput
        style={styles.search}
        placeholder="Search stocks or crypto..."
        value={search}
        onChangeText={setSearch}
      />
      <FlatList
        data={filtered}
        keyExtractor={(item) => item.symbol}
        renderItem={renderItem}
        style={styles.list}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: 'var(--color-bg-light)',
    padding: 16,
  },
```

**src/ui/PortfolioScreen.jsx**
```jsx
import React from 'react';
import { View, Text, FlatList, StyleSheet } from 'react-native';
import './Theme.css';

const dummyPortfolio = {
  cash: 12450.75,
  positions: [
    { symbol: 'AAPL', qty: 30, avgPrice: 150.2, lastPrice: 172.34 },
    { symbol: 'TSLA', qty: 10, avgPrice: 200.0, lastPrice: 215.12 },
    { symbol: 'BTCUSD', qty: 0.5, avgPrice: 30000, lastPrice: 31745.6 },
  ],
};

export default function PortfolioScreen() {
  const totalValue = dummyPortfolio.positions.reduce(
    (sum, p) => sum + p.qty * p.lastPrice,
    dummyPortfolio.cash
  );

  const renderItem = ({ item }) => {
    const marketValue = item.qty * item.lastPrice;
    const profit = marketValue - item.qty * item.avgPrice;
    return (
      <View style={styles.positionRow}>
        <Text style={styles.symbol}>{item.symbol}</Text>
        <Text style={styles.qty}>x{item.qty}</Text>
        <Text style={styles.value}>${marketValue.toFixed(2)}</Text>
        <Text style={[styles.profit, profit >= 0 ? styles.profitPos : styles.profitNeg]}>
          {profit >= 0 ? '+' : ''}{profit.toFixed(2)}
        </Text>
      </View>
    );
  };

  return (
    <View style={styles.container}>
      <Text style={styles.header}>Portfolio Overview</Text>
      <Text style={styles.cash}>Cash Balance: ${dummyPortfolio.cash.toLocaleString()}</Text>
      <FlatList
        data={dummyPortfolio.positions}
        keyExtractor={(item) => item.symbol}
        renderItem={renderItem}
        style={styles.list}
      />
      <Text style={styles.total}>Total Net Worth: ${totalValue.toLocaleString()}</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: 'var(--color-bg-light)',
    padding: 16,
  },
  header: {
    fontSize: 22,
    fontWeight: '700',
    color: 'var(--color-primary-blue)',
    marginBottom: 12,
    textAlign: 'center',
  },
  cash: {
    fontSize: 16,
    marginBottom: 12,
    color: 'var(--color-text-primary)',
  },
  list: { flexGrow: 0 },
  positionRow: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    paddingVertical: 8,
    borderBottomWidth: 1,
    borderColor: 'var(--color-border)',
  },
  symbol: { fontWeight: '600', color: 'var(--color-primary-blue)' },
  qty: { color: 'var(--color-text-secondary)' },
  value: { fontWeight: '600' },
  profit: { fontWeight: '600' },
  profitPos: { color: 'var(--color-success)' },
  profitNeg: { color: 'var(--color-error)' },
  total: {
    marginTop: 16,
    fontSize: 18,
    fontWeight: '700',
    textAlign: 'center',
    color: 'var(--color-primary-gold)',
  },
});
```

**Explanation**

- All components are written in **React‑Native JSX** and use **StyleSheet** for styling.
- Global theme variables are defined in `Theme.css` and referenced via CSS custom properties (`var(--color-…)`) which works with React‑Native Web and Expo’s support for CSS.
- Each screen follows the UI/UX spec:
  - **HomeScreen** – shows a leaderboard ranking list and a “Start Trading” button.
  - **MarketScreen** – searchable list of stocks/crypto with price preview.
  - **StockDetail** – placeholder candlestick chart with **Buy** and **Sell** buttons (colored blue/red).
  - **PortfolioScreen** – displays cash balance, positions, profit/loss, and total net worth.
- The theme provides blue, red, and gold primary colors as required.

All files are located under `src/ui/` as requested.
