```jsx
import React, { useState, useEffect } from 'react';
import {
  View,
  Text,
  StyleSheet,
  TextInput,
  FlatList,
  TouchableOpacity,
} from 'react-native';
import { useNavigation } from '@react-navigation/native';
import { Theme } from './Theme.css';

const mockStocks = [
  { symbol: 'AAPL', name: 'Apple Inc.', price: 172.3, change: 1.2 },
  { symbol: 'GOOGL', name: 'Alphabet Inc.', price: 124.5, change: -0.8 },
  { symbol: 'TSLA', name: 'Tesla, Inc.', price: 215.7, change: 2.4 },
  // ...more mock data
];

export default function MarketScreen() {
  const navigation = useNavigation();
  const [query, setQuery] = useState('');
  const [filterUp, setFilterUp] = useState(null); // true=up, false=down, null=all
  const [data, setData] = useState(mockStocks);

  useEffect(() => {
    let filtered = mockStocks.filter(
      (s) =>
        s.symbol.includes(query.toUpperCase()) ||
        s.name.toLowerCase().includes(query.toLowerCase())
    );
    if (filterUp !== null) {
      filtered = filtered.filter((s) => (filterUp ? s.change > 0 : s.change < 0));
    }
    setData(filtered);
  }, [query, filterUp]);

  const renderItem = ({ item }) => (
    <TouchableOpacity
      style={styles.row}
      onPress={() => navigation.navigate('StockDetail', { symbol: item.symbol })}
    >
      <View style={styles.info}>
        <Text style={styles.symbol}>{item.symbol}</Text>
        <Text style={styles.name}>{item.name}</Text>
      </View>
      <View style={styles.priceBox}>
        <Text style={styles.price}>${item.price.toFixed(2)}</Text>
        <Text
          style={[
            styles.change,
            { color: item.change >= 0 ? Theme.success : Theme.danger },
          ]}
        >
          {item.change > 0 ? `+${item.change}%` : `${item.change}%`}
        </Text>
      </View>
    </TouchableOpacity>
  );

  return (
    <View style={styles.container}>
      <Text style={styles.title}>市場</Text>
      <TextInput
        style={styles.search}
        placeholder="搜尋股票代碼或名稱"
        placeholderTextColor={Theme.placeholder}
        value={query}
        onChangeText={setQuery}
      />
      <View style={styles.filterBar}>
        <TouchableOpacity
          style={[
            styles.filterBtn,
            filterUp === true && styles.filterActive,
          ]}
          onPress={() => setFilterUp(filterUp === true ? null : true)}
        >
          <Text style={styles.filterText}>上漲</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[
            styles.filterBtn,
            filterUp === false && styles.filterActive,
          ]}
          onPress={() => setFilterUp(filterUp === false ? null : false)}
        >
          <Text style={styles.filterText}>下跌</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.filterBtn, filterUp === null && styles.filterActive]}
          onPress={() => setFilterUp(null)}
        >
          <Text style={styles.filterText}>全部</Text>
        </TouchableOpacity>
      </View>
      <FlatList
        data={data}
        keyExtractor={(item) => item.symbol}
        renderItem={renderItem}
        style={styles.list}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: Theme.background,
    padding: 16,
  },
  title: {
    fontSize: 22,
    color: Theme.primary,
    fontWeight: 'bold',
    marginBottom: 12,
  },
  search: {
    height: 40,
    borderColor: Theme.divider,
    borderWidth: 1,
    borderRadius: 8,
    paddingHorizontal: 10,
    color: Theme.text,
    marginBottom: 12,
  },
  filterBar: {
    flexDirection: 'row',
    marginBottom: 12,
    justifyContent: 'space-around',
  },
  filterBtn: {
    paddingVertical: 6,
    paddingHorizontal: 12,
    borderRadius: 6,
    backgroundColor: Theme.buttonBg,
  },
  filterActive: {
    backgroundColor: Theme.accent,
  },
  filterText: {
    color: Theme.buttonText,
  },
  list: { flex: 1 },
  row: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    paddingVertical: 10,
    borderBottomColor: Theme.divider,
    borderBottomWidth: 1,
  },
  info: { flexDirection: 'column' },
  symbol: { color: Theme.text, fontWeight: '600' },
  name: { color: Theme.placeholder, fontSize: 12 },
  priceBox: { alignItems: 'flex-end' },
  price: { color: Theme.text, fontWeight: '600' },
  change: { fontSize: 12 },
});
```

```jsx
  const chartData = {

            styles.tradeBtn,

        >
          <Text style={styles.tradeBtnText}>買入</Text>

        <TouchableOpacity style={styles.executeBtn} onPress={handleTrade}>
          <Text style={styles.executeBtnText}>執行</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}
```

```jsx
import React, { useState, useEffect } from 'react';
import { View, Text, StyleSheet, FlatList, ActivityIndicator } from 'react-native';
import { Theme } from './Theme.css';
import { BarChart } from 'react-native-chart-kit';
import { Dimensions } from 'react-native';

const screenWidth = Dimensions.get('window').width;

export default function PortfolioScreen() {
  const [loading, setLoading] = useState(true);
  const [portfolio, setPortfolio] = useState(null);

  // Mock load
  useEffect(() => {
    setTimeout(() => {
      setPortfolio({
        name: '主力投資組合',
        total_value: 45230.75,
        holdings: [
          { instrument_id: 'AAPL', quantity: 30, avg_price: 150, last_updated: '2024-09-01' },
          { instrument_id: 'TSLA', quantity: 10, avg_price: 210, last_updated: '2024-09-01' },
          { instrument_id: 'GOOGL', quantity: 5, avg_price: 120, last_updated: '2024-09-01' },
        ],
      });
      setLoading(false);
    }, 800);
  }, []);

  if (loading) {
    return (
      <View style={styles.center}>
        <ActivityIndicator size="large" color={Theme.accent} />
      </View>
    );
  }

  const chartData = {
    labels: portfolio.holdings.map((h) => h.instrument_id),
    datasets: [
      {
        data: portfolio.holdings.map(
          (h) => h.quantity * h.avg_price
        ),
      },
    ],
  };

  const renderHolding = ({ item }) => (
    <View style={styles.holdRow}>
      <Text style={styles.holdSymbol}>{item.instrument_id}</Text>
      <Text style={styles.holdQty}>Qty: {item.quantity}</Text>
      <Text style={styles.holdValue}>≈ ${(item.quantity * item.avg_price).toFixed(2)}</Text>
    </View>
  );

  return (
    <View style={styles.container}>
      <Text style={styles.title}>投資組合 - {portfolio.name}</Text>
      <Text style={styles.total}>總資產: ${portfolio.total_value.toLocaleString()}</Text>

      <BarChart
        data={chartData}
        width={screenWidth - 32}
        height={220}
        yAxisLabel="$"
        chartConfig={{
          backgroundColor: Theme.background,
          backgroundGradientFrom: Theme.background,
          backgroundGradientTo: Theme.background,
          decimalPlaces: 0,
          color: (opacity = 1) => Theme.accent,
          labelColor: (opacity = 1) => Theme.placeholder,
          style: { borderRadius: 16 },
        }}
        style={styles.chart}
      />

      <Text style={styles.subTitle}>持股明細</Text>
      <FlatList
        data={portfolio.holdings}
        keyExtractor={(item) => item.instrument_id}
        renderItem={renderHolding}
        style={styles.list}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: Theme.background, padding: 16 },
  center: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  title: { fontSize: 22, color: Theme.primary, fontWeight: 'bold', marginBottom: 4 },
  total: { fontSize: 18, color: Theme.text, marginBottom: 12 },
  chart: { marginVertical: 12, borderRadius: 16 },
  subTitle: { fontSize: 18, color: Theme.primary, marginTop: 12, marginBottom: 6 },

    borderBottomWidth: 1,
```
